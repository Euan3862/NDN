name: Track Overleaf Word Count

on:
  push:
    branches: [ "main" ]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  track-word-count:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Install Official TeXcount
        run: |
          sudo apt-get update
          sudo apt-get install -y texlive-extra-utils

      - name: Calculate Word Count
        id: count
        run: |
          # The -merge flag ensures \input and \include files are counted
          # awk '{print $1}' ensures we ONLY grab the primary word count number safely
          RAW_COUNT=$(texcount -merge -brief -sum main.tex 2>/dev/null | awk '{print $1}')
          
          # Fallback to 0 if empty or not a number
          if ! [[ "$RAW_COUNT" =~ ^[0-9]+$ ]]; then
            RAW_COUNT=0
          fi
          
          echo "word_count=$RAW_COUNT" >> $GITHUB_OUTPUT
          echo "Current word count: $RAW_COUNT"

      - name: Update Word History CSV
        run: |
          DATE=$(date '+%Y-%m-%d %H:%M')
          COUNT="${{ steps.count.outputs.word_count }}"
          
          if [ ! -f "word_history.csv" ]; then
            echo "date,count" > word_history.csv
          fi
          
          echo "$DATE,$COUNT" >> word_history.csv

      - name: Commit and push changes to repo
        uses: stefanzweifel/git-auto-commit-action@v5
        with:
          commit_message: "Auto-update word count history [skip ci]"
          file_pattern: word_history.csv
